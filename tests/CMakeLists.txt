# Declares every test of tg-cli and where it is built: build/tests/<category>/bin/<name>.

# policies/testing-model.md, "How a test is declared, built and found":
# every test is one project_test(<category> <name> ...) line below, and the file
# build/tests/<category>/bin/<name> is the test, whether it is a binary or a
# script.  unit/check_tests.sh fails on a declared test with no file in bin/
# and on a file in bin/ that is not declared.
#
# Programs the tests drive (harnesses, mock servers, the product binaries) are
# not tests and are not in bin/.

set(TG_TEST_BIN_ROOT "${CMAKE_CURRENT_BINARY_DIR}" CACHE PATH
    "Directory that receives <category>/bin/<name> for every declared test")
set(TG_TEST_TIMEOUT 600)

# project_test(<category> <name> TARGET <target> [TIMEOUT <s>] [NO_CTEST])
# project_test(<category> <name> SCRIPT <file> [ARGS <arg>...] [TIMEOUT <s>])
#
# TARGET: an executable defined by the category's own CMakeLists.txt is
# placed in bin/ under <name>.  SCRIPT: a small wrapper in bin/ runs the script
# with its arguments, so that everything in bin/ runs by its path alone.
# NO_CTEST: built and declared, but not run by `ctest` (it needs the outside
# world, e.g. the integration runner needs a real Telegram test DC).
function(project_test _cat _name)
    cmake_parse_arguments(T "NO_CTEST" "TARGET;SCRIPT;TIMEOUT" "ARGS" ${ARGN})
    set(_bin "${TG_TEST_BIN_ROOT}/${_cat}/bin")
    if(NOT T_TIMEOUT)
        set(T_TIMEOUT ${TG_TEST_TIMEOUT})
    endif()
    if(T_TARGET)
        get_target_property(_wd ${T_TARGET} BINARY_DIR)
        set_target_properties(${T_TARGET} PROPERTIES
            RUNTIME_OUTPUT_DIRECTORY "${_bin}"
            OUTPUT_NAME "${_name}")
        set(_cmd "${_bin}/${_name}")
        set(_workdir "${_wd}")
    elseif(T_SCRIPT)
        if(IS_ABSOLUTE "${T_SCRIPT}")
            set(_script "${T_SCRIPT}")
        else()
            set(_script "${CMAKE_CURRENT_SOURCE_DIR}/${T_SCRIPT}")
        endif()
        set(_line "'${_script}'")
        foreach(_a IN LISTS T_ARGS)
            string(APPEND _line " '${_a}'")
        endforeach()
        file(GENERATE OUTPUT "${_bin}/${_name}"
             CONTENT "#!/bin/sh\n# Generated by tests/CMakeLists.txt: runs ${T_SCRIPT}.\nexec ${_line} \"$@\"\n"
             FILE_PERMISSIONS OWNER_READ OWNER_WRITE OWNER_EXECUTE
                              GROUP_READ GROUP_EXECUTE WORLD_READ WORLD_EXECUTE)
        set(_cmd "${_bin}/${_name}")
        set(_workdir "${CMAKE_CURRENT_BINARY_DIR}")
    else()
        message(FATAL_ERROR "project_test(${_cat} ${_name}): TARGET or SCRIPT is required")
    endif()
    if(NOT T_NO_CTEST)
        add_test(NAME ${_cat}.${_name} COMMAND "${_cmd}" WORKING_DIRECTORY "${_workdir}")
        set_tests_properties(${_cat}.${_name} PROPERTIES LABELS ${_cat} TIMEOUT ${T_TIMEOUT})
    endif()
endfunction()

add_subdirectory(unit)
add_subdirectory(functional)
add_subdirectory(integration)
if(UNIX)
    add_subdirectory(pty)
endif()

# --- unit ------------------------------------------------------------------
project_test(unit        test-runner                  TARGET test-runner)
project_test(unit        check-skill   SCRIPT unit/check_skill.sh ARGS ${CMAKE_SOURCE_DIR} TIMEOUT 60)
project_test(unit        check-license SCRIPT unit/check_license.sh ARGS ${CMAKE_SOURCE_DIR} TIMEOUT 60)
project_test(unit        check-headers SCRIPT unit/check_headers.sh ARGS ${CMAKE_SOURCE_DIR} TIMEOUT 60)
project_test(unit        check-private-names SCRIPT unit/check_private_names.sh ARGS ${CMAKE_SOURCE_DIR} TIMEOUT 60)
# The workspace libraries are linked statically: no libtuikit.so or
# libptytest.so among the dependencies of a binary.
project_test(unit        check-static-libs SCRIPT ${CMAKE_SOURCE_DIR}/ci/check-static-libs.sh
             ARGS $<TARGET_FILE:tg-cli-ro> $<TARGET_FILE:tg-tui> $<TARGET_FILE:tg-cli> TIMEOUT 60)
project_test(unit        check-static-libs-selftest SCRIPT ci/check_static_libs_test.sh
             ARGS ${CMAKE_SOURCE_DIR} ${CMAKE_BINARY_DIR}/tests/ci TIMEOUT 60)
# The Debian package builds with CPack and carries version, XB- fields,
# shlibdeps-derived dependencies and the binaries (packaging model).
project_test(unit        timing-selftest SCRIPT ci/timing_test.sh
             ARGS ${CMAKE_SOURCE_DIR} ${CMAKE_BINARY_DIR}/tests/ci TIMEOUT 60)
project_test(unit        check-package SCRIPT unit/check_package.sh
             ARGS ${CMAKE_BINARY_DIR} ${CMAKE_BINARY_DIR}/tests/package TIMEOUT 120)
# Every declared test is in bin/, and nothing else is.
project_test(unit        check-tests   SCRIPT unit/check_tests.sh
             ARGS ${CMAKE_CURRENT_SOURCE_DIR}/CMakeLists.txt ${TG_TEST_BIN_ROOT} TIMEOUT 60)
